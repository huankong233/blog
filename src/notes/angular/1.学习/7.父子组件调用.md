在一个组件中使用另外一个组件，需要使用 `ts` 先导入, 然后在 `@Component` 中的 `imports` 中注册, 最后在 `html` 中使用。

```ts
import { Component, signal } from '@angular/core'
import { ChildComponent } from './child.ts'

@Component({
  selector: 'app-root',
  styleUrl: './app.css',
  template: `
    <child-component />
  `,
  imports: [ChildComponent],
})
export class App {
  inputText = signal('')
}
```
